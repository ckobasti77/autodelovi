export const dispatch = internalMutation({
  args: { eventId: v.id("domainEvents") },
  returns: v.null(),
  handler: async (ctx, { eventId }) => {
    const event = await ctx.db.get("domainEvents", eventId);
    if (!event || event.status === "done" || event.status === "dead") return null;
    const attempts = event.attempts + 1;
    try {
      await processEvent(ctx, event);
      await ctx.db.patch("domainEvents", eventId, {
        status: "done",
        attempts,
        processedAt: Date.now(),
        lastError: undefined,
      });
    } catch (error) {
      // Deliveries written before the failure stay (idempotent); the event retries later.
      const message = error instanceof Error ? error.message : String(error);
      const dead = attempts >= MAX_ATTEMPTS;
      await ctx.db.patch("domainEvents", eventId, {
        status: dead ? "dead" : "failed",
        attempts,
        lastError: message.slice(0, 500),
        nextAttemptAt: Date.now() + retryDelayMs(attempts),
      });
    }
    return null;
  },
});

## Docs promise (docs/NOTIFICATIONS.md:10)
10:`emitEvent(ctx, type, payload, dedupeKey)` inserts `domainEvents` (pending) **in the same transaction** as the business write and schedules `events.dispatch` with `runAfter(0)`. The dispatcher resolves recipients (settings + staff preferences), writes `notificationDeliveries` rows (unique per event × channel × recipient — idempotent), creates in-app `notifications`, and enqueues e-mails (`emails.render` → `emails.send`), CAPI (`capi.send`) and cache revalidation (`revalidate.notify`) through the `externalIo` workpool. Failures retry with exponential backoff (1 min, 5 min, 30 min, 2 h, 12 h) → `dead` after 5 attempts (visible in admin). `events.sweep` (every minute) re-dispatches stuck `pending`/`failed` events whose `nextAttemptAt` passed.
